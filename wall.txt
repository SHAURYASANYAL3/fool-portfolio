Dynamic Multi-NIC Chunk Distribution and Tail Racing Engine
Schedules byte-ranges across heterogeneous network links with dynamic work-stealing
and latency-adaptive tail-end chunk racing.
"""
from __future__ import annotations

CHUNK_STATUS_PENDING = "pending"
CHUNK_STATUS_DOWNLOADING = "downloading"
CHUNK_STATUS_COMPLETED = "completed"
CHUNK_STATUS_FAILED = "failed"
@dataclass
class Chunk:
    id: int
    start: int
    end: int
    downloaded: int = 0
    status: str = CHUNK_STATUS_PENDING
    assigned_adapter: Optional[str] = None
    assigned_time: float = 0.0
    completed_time: float = 0.0
    retries: int = 0
    @property
    def total_bytes(self) -> int:
        return self.end - self.start + 1
    @property
    def remaining_bytes(self) -> int:
        return max(0, self.total_bytes - self.downloaded)
    @property
    def is_finished(self) -> bool:
        return self.status == CHUNK_STATUS_COMPLETED or (self.downloaded >= self.total_bytes)
class ChunkEngine:
    """Manages chunk partitioning, dynamic allocation, and speculative racing."""
    def __init__(self, total_size: int, chunk_size: Optional[int] = None):
        self.total_size = total_size
        self.chunk_size = chunk_size or self._calculate_optimal_chunk_size(total_size)
        self.chunks: List[Chunk] = []
        self.lock = Lock()
        self._init_chunks()
    def _calculate_optimal_chunk_size(self, total_bytes: int) -> int:
        # Scale chunk size dynamically to balance multi-NIC parallelism and HTTP overhead
        if total_bytes < 5 * 1024 * 1024:          # < 5 MB: 512 KB chunks
            return 512 * 1024
        elif total_bytes < 50 * 1024 * 1024:       # < 50 MB: 2 MB chunks
            return 2 * 1024 * 1024
        elif total_bytes < 500 * 1024 * 1024:      # < 500 MB: 8 MB chunks
            return 8 * 1024 * 1024
        elif total_bytes < 5 * 1024 * 1024 * 1024: # < 5 GB: 16 MB chunks
            return 16 * 1024 * 1024
        else:                                      # >= 5 GB: 32 MB chunks
            return 32 * 1024 * 1024
    def _init_chunks(self) -> None:
        if self.total_size <= 0:
            return
        chunk_id = 0
        current = 0
        while current < self.total_size:
            end = min(current + self.chunk_size - 1, self.total_size - 1)

class SystemOptimizer:
    """Diagnoses and tunes operating system networking for simultaneous multi-WAN links."""
    @staticmethod
    def diagnose() -> Dict[str, any]:
        """Runs a complete dual-NIC diagnostic scan."""
        report = {
            "platform": sys.platform,
            "can_dual_bond": True,
            "issues": [],
            "recommendations": [],
            "details": {},
        }
        if sys.platform != "win32":
            report["details"]["status"] = "Non-Windows OS; socket-level device binding is fully supported."
            return report
        # 1. Check Windows Connection Manager fMinimizeConnections
        f_min_val = SystemOptimizer._check_windows_minimize_connections()
        report["details"]["fMinimizeConnections"] = f_min_val
        if f_min_val == 1 or f_min_val == 3:
            report["issues"].append(
                "Windows Connection Manager is configured to minimize simultaneous connections "
                "(fMinimizeConnections = 1/3). Windows may disconnect Wi-Fi when Ethernet is plugged in."
            )
            report["recommendations"].append(
                "Set 'Minimize the number of simultaneous connections' to Disabled in Group Policy, "
                "or run 'aetherbond doctor --fix'."
            )
            report["can_dual_bond"] = False
        # 2. Check Weak Host routing configuration
        weak_host_info = SystemOptimizer._check_weak_host_config()
        report["details"]["weak_host"] = weak_host_info
        # 3. Check active adapters
        from aetherbond.core.adapter import AdapterManager
        mgr = AdapterManager()
        adapters = mgr.get_adapters(force_refresh=True)
        report["details"]["adapters"] = [a.to_dict() for a in adapters]
        active_count = sum(1 for a in adapters if a.status.lower() == "up" and a.ipv4 and not a.ipv4.startswith("169.254"))
        if active_count < 2:
            report["recommendations"].append(
                f"Currently only {active_count} active internet adapter detected. Connect Wi-Fi and Ethernet (or Phone USB Tethering) "
                "simultaneously to achieve combined dual-link speed."
            )
        return report
    @staticmethod
    def _check_windows_minimize_connections() -> Optional[int]:
        try:
            import winreg
            key = winreg.OpenKey(
                winreg.HKEY_LOCAL_MACHINE,
                r"SOFTWARE\Policies\Microsoft\Windows\WcmSvc\GroupPolicy",
                0,
                winreg.KEY_READ,
            )
            val, _ = winreg.QueryValueEx(key, "fMinimizeConnections")
            winreg.CloseKey(key)

def is_disposable(domain: str) -> bool:
    """Checks if the given domain belongs to a disposable/burner email service."""
    domain = domain.strip().lower()
    if domain in DISPOSABLE_DOMAINS:
        return True
    # Subdomain check
    for disp in DISPOSABLE_DOMAINS:
        if domain.endswith("." + disp):
            return True
    return False
